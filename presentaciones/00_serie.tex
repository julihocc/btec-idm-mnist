\begin{frame}{La pregunta de toda la serie}
  En las cuatro partes la pregunta es la misma:

  \begin{center}
    \Large \emph{¿qué hay en esta imagen?}
  \end{center}

  Cambia \textbf{qué} responde el programa y \textbf{con qué herramienta}.
\end{frame}

\begin{frame}{Cuatro partes, un orden}
  \begin{enumerate}
    \item \textbf{Parte 1} --- \libreta{01\_taller\_mnist.ipynb}\\
          la máquina \textbf{aprende} a leer dígitos (Keras) y \textbf{ejecuta} esa red con OpenCV (\texttt{cv2.dnn})
    \item \textbf{Parte 2} --- \libreta{02\_taller\_qr.ipynb}\\
          visión en \textbf{software}: generar y leer un QR, usarlo en una mini-app
    \item \textbf{Parte 3} --- \libreta{03\_taller\_aruco.ipynb}\\
          visión en \textbf{robótica}: marcadores ArUco (ID + posición)
    \item \textbf{Parte 4} --- \libreta{04\_taller\_moda.ipynb}\\
          \textbf{reto} de visión en moda: Fashion-MNIST + OpenCV
  \end{enumerate}
\end{frame}

\begin{frame}{Cómo se diferencian}
  \small
  \begin{tabular}{@{}llp{4.6cm}@{}}
    \toprule
    Parte & Respuesta & Herramienta \\
    \midrule
    1 & un dígito 0--9 & red que \textbf{aprendiste} + \texttt{cv2.dnn} \\
    2 & un texto (SKU, URL) & decoder QR de OpenCV \\
    3 & un ID de estación & detector ArUco de OpenCV \\
    4 & calzado o no & \textbf{tu} regla OpenCV (contornos) \\
    \bottomrule
  \end{tabular}

  \vspace{1em}
  Hazlas \textbf{en ese orden}. OpenCV entra al final de la Parte~1; las demás lo reutilizan.
\end{frame}

\begin{frame}{TACO, en cada libreta}
  Cada paso de las notebooks sigue el mismo molde:

  \begin{description}
    \item[Tarea] qué hay que hacer
    \item[Enfoque] por qué / cómo
    \item[Código] una celda corta
    \item[Resultado] qué deberías ver
  \end{description}

  Estas diapositivas acompañan; el trabajo está en las libretas.
\end{frame}

\begin{frame}{Parte 1 --- dígitos}
  \textbf{Objetivo:} $>$95\,\% en prueba (corrida guardada: 97{,}08\,\%).

  \begin{itemize}
    \item Dataset MNIST (60{,}000 + 10{,}000, $28\times28$)
    \item MLP: 784 $\to$ 128 ReLU $\to$ dropout 20\,\% $\to$ 10 softmax
    \item 5 épocas, Adam, \texttt{sparse\_categorical\_crossentropy}
    \item Paso 8: la \textbf{misma} red en OpenCV (\texttt{TFLite} + \texttt{cv2.dnn})
  \end{itemize}

  Acompañamiento: \libreta{01\_mnist.pdf}.
\end{frame}

\begin{frame}{Parte 2 --- QR}
  Un QR no es un dibujo: es un \textbf{string} dentro de una imagen.

  \begin{itemize}
    \item Generar \texttt{SKU-1042} con \texttt{QRCodeEncoder}
    \item \texttt{detectAndDecode}: esquinas + texto
    \item Mini-app: catálogo (precio, stock) y un código inexistente
    \item El error de ``fuera del catálogo'' es de \textbf{negocio}, no de visión
  \end{itemize}

  Acompañamiento: \libreta{02\_qr.pdf}.
\end{frame}

\begin{frame}{Parte 3 --- ArUco}
  Un robot no lee dígitos a mano: pega un marcador diseñado.

  \begin{itemize}
    \item Diccionario \texttt{DICT\_4X4\_50}
    \item ID + 4 esquinas; el centro es la posición en la foto
    \item Margen blanco obligatorio
    \item Tres estaciones: Entrada (3), Almacén (7), Salida (12)
  \end{itemize}

  Acompañamiento: \libreta{03\_aruco.pdf}.
\end{frame}

\begin{frame}{Parte 4 --- reto de moda}
  Dataset abierto \textbf{Fashion-MNIST} (Zalando). Keras solo \emph{carga}; OpenCV \emph{mide}.

  \begin{itemize}
    \item 10 prendas $28\times28$
    \item Umbral, contornos, caja, aspecto
    \item Completar \texttt{es\_calzado}: sandalia / zapatilla / bota vs el resto
    \item Meta: $>$80\,\% en 2{,}000 de prueba (el brillo medio rinde $\sim$43\,\%)
    \item Sin entrenar y sin \texttt{model.predict}
  \end{itemize}

  Acompañamiento: \libreta{04\_moda.pdf}.
\end{frame}

\begin{frame}{Cuándo usar cada herramienta}
  \begin{description}
    \item[Red neuronal] el patrón es sucio o variable (escritura, ropa ``a mano'')
    \item[QR] tú eliges el texto (SKU, URL, ticket) --- apps e inventario
    \item[ArUco] etiquetas para robots: ID + dónde en la imagen
    \item[Regla OpenCV] forma medible (contorno, aspecto) sin volver a entrenar
  \end{description}
\end{frame}
