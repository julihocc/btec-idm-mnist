\begin{frame}{Objetivo de esta parte}
  Una red neuronal simple que clasifique más del \textbf{95\,\%} de los dígitos de prueba.

  Al final, esa \textbf{misma} red corre en OpenCV (\texttt{cv2.dnn}).

  \vspace{1em}
  Pregunta de la serie: \pregunta
\end{frame}

\begin{frame}{Dataset MNIST}
  \begin{itemize}
    \item 70{,}000 imágenes de dígitos manuscritos 0--9
    \item 60{,}000 para entrenar, 10{,}000 para probar
    \item Cada una: $28\times 28$ píxeles en escala de grises (0--255)
  \end{itemize}

  Keras lo descarga a \texttt{\string~/.keras/datasets/} la primera vez.
\end{frame}

\begin{frame}{Los 8 pasos (TACO)}
  \begin{enumerate}
    \item Preparar las herramientas (TensorFlow / Keras, NumPy, Matplotlib)
    \item Cargar MNIST
    \item Normalizar a $[0,1]$ y aplanar $28\times28 \to 784$
    \item Diseñar el MLP
    \item Entrenar 5 épocas
    \item Evaluar en prueba y ver curvas
    \item Mirar predicciones
    \item Exportar a TFLite y clasificar con \texttt{cv2.dnn}
  \end{enumerate}
\end{frame}

\begin{frame}{Arquitectura}
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Capa & Unidades & Activación & Parámetros \\
    \midrule
    Entrada & 784 & --- & 0 \\
    Densa & 128 & ReLU & 100{,}480 \\
    Dropout & 20\,\% & --- & 0 \\
    Densa & 10 & Softmax & 1{,}290 \\
    \midrule
    \textbf{Total} & & & \textbf{101{,}770} \\
    \bottomrule
  \end{tabular}

  \vspace{0.8em}
  Pérdida: \texttt{sparse\_categorical\_crossentropy} (la etiqueta es un entero 0--9).\\
  Optimizador: Adam. Lote: 128. Validación: último 10\,\% (Keras no lo baraja).
\end{frame}

\begin{frame}{Entrenar y evaluar}
  \begin{itemize}
    \item 5 épocas; precisión de entrenamiento $\sim$88\,\% $\to$ $\sim$97\,\%
    \item Validación por encima del entrenamiento: Dropout está trabajando
    \item \textbf{Prueba:} 97{,}08\,\% (9{,}708 / 10{,}000), pérdida 0{,}0948
    \item Una muestra de 16 imágenes \emph{no} sustituye ese 97\,\%
  \end{itemize}

  Objetivo de la parte: $>$95\,\%. Cumplido.
\end{frame}

\begin{frame}{Paso 8: Keras $\to$ OpenCV}
  OpenCV 5 no trae el kNN clásico (\texttt{cv2.ml}). Sí trae \texttt{cv2.dnn}.

  \begin{enumerate}
    \item Exportar el MLP a \textbf{TFLite} (viene con TensorFlow)
    \item \texttt{cv2.dnn.readNetFromTFLite}
    \item Clasificar las mismas 16 imágenes del paso 7
  \end{enumerate}

  En la corrida guardada: \textbf{16/16} coincidencias Keras vs OpenCV\\
  (incluso el error compartido: un 5 leído como 6).

  A partir de aquí \texttt{cv2} ya está en el taller.
\end{frame}

\begin{frame}{Qué no hacer en esta parte}
  \begin{itemize}
    \item No es un CNN (eso queda como extra)
    \item No hay webcam
    \item OpenCV \textbf{no entrena}: solo \textbf{corre} el modelo
  \end{itemize}
\end{frame}
